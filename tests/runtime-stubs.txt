class WP_Error {
	private $code;
	private $message;
	private $errors = array();
	public function __construct( $code = '', $message = '' ) { $this->code = $code; $this->message = $message; }
	public function get_error_code() { return $this->code ?: ( $this->errors ? array_key_first( $this->errors ) : '' ); }
	public function get_error_message() { if ( $this->message ) return $this->message; $code = $this->get_error_code(); return $code ? $this->errors[ $code ][0] : ''; }
	public function add( $code, $message ) { $this->errors[ $code ][] = $message; }
	public function has_errors() { return (bool) $this->errors; }
}
